#include <stdio.h>
#include <pthread.h>
#include <semaphore.h>
#include <unistd.h>
#define SIZE 5
int buffer[SIZE];
int in = 0;
int out = 0;
sem_t empty;
sem_t full;
pthread_mutex_t mutex;
void *producer(void *arg)
{
int item;
for(item = 1; item <= 10; item++)
{
sem_wait(&empty);
pthread_mutex_lock(&mutex);
buffer[in] = item;
printf("Producer produced: %d\n", item);
in = (in + 1) % SIZE;
pthread_mutex_unlock(&mutex);
sem_post(&full);
sleep(1);
}
return NULL;
}
void *consumer(void *arg)
{
int item;
for(int i = 1; i <= 10; i++)
{
sem_wait(&full);
pthread_mutex_lock(&mutex);
item = buffer[out];
printf("Consumer consumed: %d\n", item);
out = (out + 1) % SIZE;
pthread_mutex_unlock(&mutex);
sem_post(&empty);
sleep(1);
}
return NULL;
}
int main()
{
pthread_t producerThread;
pthread_t consumerThread;
sem_init(&empty, 0, SIZE);
sem_init(&full, 0, 0);
pthread_mutex_init(&mutex, NULL);
pthread_create(&producerThread, NULL, producer, NULL);
pthread_create(&consumerThread, NULL, consumer, NULL);
pthread_join(producerThread, NULL);
pthread_join(consumerThread, NULL);
sem_destroy(&empty);
sem_destroy(&full);
pthread_mutex_destroy(&mutex);
return 0;
}




Aim
To implement the Producer-Consumer problem using threads, semaphores, and mutex locks in C.
Objectives
1. To understand the Producer-Consumer problem.
2. To implement multithreading using POSIX threads.
3. To use semaphores for process synchronization.
4. To use mutex locks to prevent race conditions.
Theory
- Producer: Generates items and places them into the shared buffer.
- Consumer: Removes and consumes items from the buffer.
- Semaphore: Controls access to the buffer and prevents overflow or underflow.
- Mutex: Ensures only one thread accesses the shared buffer at a time.
- pthread: Used to create and manage threads.
Algorithm
1. Start the program.
2. Initialize the buffer, semaphores, and mutex.
3. Create producer and consumer threads.
4. Producer generates items and inserts them into the buffer.
5. Consumer removes items from the buffer.
6. Use semaphores to manage empty and full buffer slots.
7. Use mutex locks for mutual exclusion.
8. Wait for both threads to complete using pthread_join().
9. Destroy semaphores and mutex.
10. Stop the program.
Conclusion
Thus, we successfully implemented the Producer-Consumer problem using threads, semaphores, and mutex locks, ensuring proper synchronization and avoiding race conditions.
